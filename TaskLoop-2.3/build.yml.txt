name: APK bauen
on:
  push:
  workflow_dispatch:
jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'
      - name: Android SDK vorbereiten
        run: |
          yes | "$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager" --licenses > /dev/null || true
          "$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager" "platforms;android-35" "build-tools;35.0.0"
      - name: APK bauen und signieren
        run: |
          mkdir -p .signing
          cp signing/release.jks signing/password .signing/
          export ANDROID_SDK_ROOT="$ANDROID_HOME"
          python3 build.py
          cp ../TaskLoop-2.3.apk ./TaskLoop-2.3.apk
      - uses: actions/upload-artifact@v4
        with:
          name: TaskLoop-APK
          path: TaskLoop-2.3.apk
